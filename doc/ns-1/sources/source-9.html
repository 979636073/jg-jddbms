314     private List<Header> setColumnInfo(List<Header> headers, String tableName, String schemaName, String databaseName) {
315         try {
316             if (StrUtil.isEmpty(tableName) || (StrUtil.isEmpty(schemaName) && StrUtil.isEmpty(databaseName))) {
317                 return headers;
318             }
319             tableName = tableName.replaceAll("\"", "")
320                     .replaceAll("`", "")
321                     .replaceAll("'", "");
322             if (tableName.indexOf(".") > -1) {
323                 tableName = StrUtil.split(tableName, ".").get(1);
324             }
325             TableQueryParam tableQueryParam = new TableQueryParam();
326             tableQueryParam.setTableName(MetaNameUtils.getMetaName(tableName));
327             tableQueryParam.setSchemaName(schemaName);
328             tableQueryParam.setDatabaseName(databaseName);
329             tableQueryParam.setRefresh(true);
330             List<TableColumn> columns = tableService.queryColumns(tableQueryParam);
331             if (CollectionUtils.isEmpty(columns)) {
332                 return headers;
333             }
334             Map<String, TableColumn> columnMap = columns.stream().collect(
335                     Collectors.toMap(TableColumn::getName, tableColumn -> tableColumn));
336             List<TableIndex> tableIndices = tableService.queryIndexes(tableQueryParam);
337             if (!CollectionUtils.isEmpty(tableIndices)) {
338                 for (TableIndex tableIndex : tableIndices) {
339                     if ("PRIMARY".equalsIgnoreCase(tableIndex.getType())) {
340                         List<TableIndexColumn> columnList = tableIndex.getColumnList();
341                         if (!CollectionUtils.isEmpty(columnList)) {
342                             for (TableIndexColumn tableIndexColumn : columnList) {
343                                 TableColumn tableColumn = columnMap.get(tableIndexColumn.getColumnName());
344                                 if (tableColumn != null) {
345                                     tableColumn.setPrimaryKey(true);
346                                 }
347                             }
348                         }
349                     }
350                 }
351             }
352             for (Header header : headers) {
353                 TableColumn tableColumn = columnMap.get(header.getName());
354                 if (tableColumn != null) {
355                     header.setPrimaryKey(tableColumn.getPrimaryKey());
356                     //解决达梦自增序列问题
357                     header.setAutoIncrement(Validator.isNotEmpty(tableColumn.getAutoIncrement()) && tableColumn.getAutoIncrement() ? 1 : 0);
358                     header.setComment(tableColumn.getComment());
359                     header.setDefaultValue(tableColumn.getDefaultValue());
360                     header.setNullable(tableColumn.getNullable());
361                     header.setColumnSize(tableColumn.getColumnSize());
362                     header.setDecimalDigits(tableColumn.getDecimalDigits());
363                 }
364             }
365 
366         } catch (Exception e) {
367             log.error("setColumnInfo error:", e);
368         }
369         return headers;
370     }
