159     private void sortSchema(List<Schema> schemas, Connection connection) {
160         if (CollectionUtils.isEmpty(schemas)) {
161             return;
162         }
163         String ulr = null;
164         try {
165             ulr = connection.getMetaData().getURL();
166         } catch (SQLException e) {
167             log.error("get url error", e);
168         }
169         // If the database name contains the name of the current database, the current database is placed in the first place
170         int num = -1;
171         for (int i = 0; i < schemas.size(); i++) {
172             String schema = schemas.get(i).getName();
173             if (StringUtils.isNotBlank(ulr) && schema != null && ulr.contains(schema)) {
174                 num = i;
175                 break;
176             }
177         }
178         if (num != -1 && num != 0) {
179             Collections.swap(schemas, num, 0);
180         }
181     }

737     private void doImport(DatabaseImportRequest request, File importFile, File logFile, DatabaseImportVo vo, String key) {
738         vo.setStatus(TaskStatusEnum.PROCESSING.name());
739         final Connection connection = Chat2DBContext.getConnection();
740         Statement statement = null;
741         try {
742             statement = connection.createStatement();
743             String logFormat = "%s|%s|%s";
744             int exported = 1;
745             int errorNum = 0;
746             int validLine = 0;
747             try (PrintWriter printWriter = new PrintWriter(logFile, StandardCharsets.UTF_8.name());
748                  BufferedReader reader = new BufferedReader(new FileReader(importFile))) {
749                 String line;
750                 StringBuilder sql = new StringBuilder();
751                 boolean insideBlockComment = false;
752                 while ((line = reader.readLine()) != null) {
753                     exported++;
754                     vo.setExported(exported);
755                     line = line.trim();
756                     // 忽略空行
757                     if (line.isEmpty()) {
758                         continue;
759                     }
760 
761                     // 处理块注释
762                     if (line.startsWith("/*")) {
763                         insideBlockComment = true;
764                         continue;
765                     }
766                     if (insideBlockComment && line.endsWith("*/")) {
767                         insideBlockComment = false;
768                         continue;
769                     }
770                     if (insideBlockComment) {
771                         continue;
772                     }
773 
774                     // 处理单行注释
775                     if (line.startsWith("--")) {
776                         continue;
777                     }
778                     // 如果不在注释或字符串中，则追加到SQL语句
779                     sql.append(line);
780                     sql.append("\n");
781                     // 检查是否结束了一个完整的SQL语句（以分号结尾）
782                     if (line.endsWith(";")) {
783                         String sqlStatement = sql.toString().trim();
784                         //校验SQL是否正确，若不正确则继续读下一行，若连续1000（这个数据按实际情况改）行都不正确，则往后走，不继续校验。
785                         final boolean sqlValid = SqlUtils.isSQLValid(sqlStatement);
786                         if (!sqlValid && validLine < 1000) {
787                             validLine++;
788                             continue;
789                         }
790                         validLine = 0;
791                         // 执行SQL语句
792                         if (StrUtil.startWithIgnoreCase(sqlStatement, "SELECT")) {
793                             // 处理查询结果
794                         } else {
795                             if (vo.getMessage().size() > 50) {
796                                 vo.getMessage().remove(0);
797                             }
798                             errorNum = executeUpdate(vo, statement, logFormat, errorNum, printWriter, sqlStatement);
799                             redisCache.setCacheObject(key, vo);
800                         }
801 
802                         sql.setLength(0); // 清空StringBuilder以便存储下一个SQL语句
803                     }
804                     if (sql.length() > 0) {
805                         String sqlStatement = sql.toString().trim();
806                         errorNum = executeUpdate(vo, statement, logFormat, errorNum, printWriter, sqlStatement);
807                     }
808 
809                 }
810             } catch (Exception e) {
811                 vo.setStatus(TaskStatusEnum.ERROR.name());
812                 vo.setMessage(Arrays.asList(e.getMessage()));
813                 redisCache.setCacheObject(key, vo);
814                 return;
815             }
816         } catch (SQLException e) {
817             vo.setStatus(TaskStatusEnum.ERROR.name());
818             vo.setMessage(Arrays.asList(e.getMessage()));
819             redisCache.setCacheObject(key, vo);
820             return;
821         } finally {
822             try {
823                 if (statement != null) {
824                     statement.close();
825                 }
826                 if (connection != null) {
827                     connection.close();
828                 }
829             } catch (SQLException e) {
830                 log.error("close connection error:{}", e);
831             }
832         }
833     }
